MLD (Modèle Logique de Données)
Key: PK = Primary Key, FK = Foreign Key, 🆕 = new table vs original
LEAVE_TYPE 🆕
id_type (INT, PK)
name (VARCHAR, UNIQUE)
default_days (INT)
DEPARTMENT
id_dept (INT, PK)
name (VARCHAR)
description (TEXT)
manager_id (INT, FK -> EMPLOYEE.id_emp)
EMPLOYEE
id_emp (INT, PK)
name (VARCHAR)
email (VARCHAR, UNIQUE)
password_hash (VARCHAR)
phone (VARCHAR)
gender (VARCHAR)
date_birth (DATE)
address (VARCHAR)
date_employment (DATE)
role (ENUM: 'Admin', 'Agent', 'Employee')
id_dept (INT, FK -> DEPARTMENT.id_dept)
supervisor_id (INT, FK -> EMPLOYEE.id_emp)
CANDIDAT
id_cand (INT, PK)
name (VARCHAR)
email (VARCHAR, UNIQUE)
date_birth (DATE)
address (VARCHAR)
date_candidature (DATE)
cv_doc_path (VARCHAR)
post_applied (VARCHAR)
id_dept (INT, FK -> DEPARTMENT.id_dept)
status (ENUM: 'Pending', 'In Progress', 'Accepted', 'Rejected')
agent_in_charge (INT, FK -> EMPLOYEE.id_emp)
ENTRETIEN
id_entretien (INT, PK)
date_heure (DATETIME)
status (ENUM: 'Scheduled', 'Completed', 'Cancelled')
result_notes (TEXT)
id_cand (INT, FK -> CANDIDAT.id_cand)
id_agent (INT, FK -> EMPLOYEE.id_emp)
CONTRACT
id_contract (INT, PK)
type (ENUM: 'CDI', 'CDD', 'Stage')
status (ENUM: 'Active', 'Terminated', 'Expired')
date_deb (DATE)
date_fin (DATE, NULLABLE)
salaire_base (DECIMAL)
id_emp (INT, FK -> EMPLOYEE.id_emp)
EVALUATION
id_eval (INT, PK)
date_eval (DATE)
score (INT)
bonus_amount (DECIMAL)
comments (TEXT)
type_eval (ENUM: 'Employee', 'Candidate')
evaluator_id (INT, FK -> EMPLOYEE.id_emp)
evaluatee_emp_id (INT, NULLABLE, FK -> EMPLOYEE.id_emp)
evaluatee_cand_id (INT, NULLABLE, FK -> CANDIDAT.id_cand)
ABSENCE
id_absence (INT, PK)
date_absence (DATE)
id_type (INT, FK -> LEAVE_TYPE.id_type)
is_justified (BOOLEAN)
justification_doc (VARCHAR, NULLABLE)
id_emp (INT, FK -> EMPLOYEE.id_emp)
recorded_by (INT, FK -> EMPLOYEE.id_emp)
conge_id (INT, NULLABLE, FK -> CONGE.id_conge)
CONGE
id_conge (INT, PK)
id_type (INT, FK -> LEAVE_TYPE.id_type)
date_deb (DATE)
date_fin (DATE)
status (ENUM: 'Pending', 'Approved', 'Rejected')
approved_by (INT, NULLABLE, FK -> EMPLOYEE.id_emp)
id_emp (INT, FK -> EMPLOYEE.id_emp)
CONGE_BALANCE 🆕
id_balance (INT, PK)
id_emp (INT, FK -> EMPLOYEE.id_emp)
id_type (INT, FK -> LEAVE_TYPE.id_type)
year (INT)
allocated (INT)
consumed (INT)
carried_over (INT)
UNIQUE (id_emp, id_type, year)
TASK
id_task (INT, PK)
name (VARCHAR)
description (TEXT)
priority (ENUM: 'Low', 'Medium', 'High')
date_deb (DATE)
date_fin (DATE)
status (ENUM: 'To Do', 'Doing', 'Done')
assigned_by (INT, FK -> EMPLOYEE.id_emp)
assigned_to (INT, FK -> EMPLOYEE.id_emp)
FORMATION
id_formation (INT, PK)
name (VARCHAR)
description (TEXT)
location (VARCHAR)
date_deb (DATE)
duration_days (INT)
instructor_id (INT, FK -> EMPLOYEE.id_emp)
PARTICIPATION_FORMATION
id_emp (INT, FK -> EMPLOYEE.id_emp)
id_formation (INT, FK -> FORMATION.id_formation)
PRIMARY KEY (id_emp, id_formation)
SALAIRE
id_salaire (INT, PK)
month_year (VARCHAR)
bonus_amount (DECIMAL)
absence_deductions (DECIMAL)
amount_final (DECIMAL)
status (ENUM: 'Generated', 'Validated', 'Paid')
id_emp (INT, FK -> EMPLOYEE.id_emp)
id_contract (INT, FK -> CONTRACT.id_contract)
NOTIFICATION 🆕
id_notif (INT, PK)
recipient_id (INT, FK -> EMPLOYEE.id_emp)
type (ENUM: 'CONTRACT_EXPIRY', 'LEAVE_PENDING', 'LEAVE_APPROVED', 'LEAVE_REJECTED', 'PAYROLL_GENERATED')
message (TEXT)
is_read (BOOLEAN, DEFAULT false)
created_at (DATETIME, DEFAULT now)
target_model (VARCHAR)
target_id (INT)
AUDIT_LOG 🆕
id_log (INT, PK)
actor_id (INT, NULLABLE, FK -> EMPLOYEE.id_emp)
action (ENUM: 'CREATE', 'UPDATE', 'DELETE', 'APPROVE', 'REJECT', 'GENERATE')
target_model (VARCHAR)
target_id (INT)
snapshot (TEXT)
timestamp (DATETIME, DEFAULT now)
